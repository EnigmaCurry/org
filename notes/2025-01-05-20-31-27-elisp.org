    "Sum the multiples of 3 or 5 below n."
    (apply #'+
           (seq-filter (lambda (x) (or (= (% x 3) 0) (= (% x 5) 0)))
                       (number-sequence 1 (1- n)))))

** Fibonacci Numbers

#+NAME: fibonacci
#+begin_src elisp :exports true :wrap stdout :linenos true
  (defun fibonacci (n)
    "Generate the fibonacci sequence below n."
    (let ((fib (lambda (a b)
                 (if (>= a n)
                     nil
                   (cons a (funcall fib b (+ a b) ))))))
      (funcall fib 1 2)))
#+end_src

** Prime Factors

#+begin_src elisp :exports true :wrap stdout :linenos true
  (defun prime-factors-of-n (n &optional divisor factors)
    "Calculate the prime factors of n"
    (cond
     ((<= n 1) (reverse factors))
     ((zerop (mod n divisor))
      (prime-factors-of-n (/ n divisor) divisor (cons divisor factors)))
     (t
      (prime-factors-of-n (1+ divisor) factors))))
    (prime-factors-of-n 56)
