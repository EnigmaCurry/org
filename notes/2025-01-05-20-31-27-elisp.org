    "Sum the multiples of 3 or 5 below n."
    (apply #'+
           (seq-filter (lambda (x) (or (= (% x 3) 0) (= (% x 5) 0)))
                       (number-sequence 1 (1- n)))))

** Fibonacci Numbers

#+NAME: fibonacci
#+begin_src elisp :exports true :linenos true
  (defun fibonacci (n)
    "Generate the fibonacci sequence below n."
    (let ((fib (lambda (a b)
                 (if (>= a n)
                     nil
                   (cons a (funcall fib b (+ a b) ))))))
      (funcall fib 1 2)))
#+end_src

** Prime Factors

#+NAME: prime-factors
#+begin_src elisp :exports true :linenos true
  (defun prime-factors (n)
  "Calculate the prime factors of n."
  (let ((factors '())
        (divisor 2))
    ;; Remove factors of 2
    (while (zerop (mod n divisor))
      (push divisor factors)
      (setq n (/ n divisor)))
    ;; Check odd divisors
    (setq divisor 3)
    (while (<= (* divisor divisor) n)
      (if (zerop (mod n divisor))
          (progn
            (push divisor factors)
            (setq n (/ n divisor)))
        (setq divisor (+ divisor 2))))
    ;; If n is still greater than 1, it is a prime number
    (if (> n 1)
        (push n factors))
    (reverse factors)))
#+end_src
